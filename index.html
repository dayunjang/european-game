<script>

    const socket = io();

    let role = '';

    let selectedBag = [];


    function joinAsStudent() {

        const name = document.getElementById('student-name').value;

        if (!name) return alert('이름 또는 모둠명을 입력해주세요!');

        role = 'student';

        socket.emit('join_game', { name: name });

        document.getElementById('login-screen').style.display = 'none';

        document.getElementById('student-screen').style.display = 'block';

    }


    function joinAsTeacher() {

        role = 'teacher';

        document.getElementById('login-screen').style.display = 'none';

        document.getElementById('teacher-screen').style.display = 'block';

    }


    function setCity(index) {

        socket.emit('teacher_command', { action: 'change_city', index: index });

    }


    function setStep(step) {

        socket.emit('teacher_command', { action: 'next_step', step: step });

    }


    socket.on('update_state', (state) => {

        if (role === 'teacher') {

            const listEl = document.getElementById('student-list');

            listEl.innerHTML = '';

            const ids = Object.keys(state.students);

            if (ids.length === 0) {

                listEl.innerHTML = '<li style="color: #94a3b8; text-align: center; border: none;">접속한 학생이 없습니다.</li>';

            } else {

                ids.forEach(id => {

                    const s = state.students[id];

                    listEl.innerHTML += `<li>👤 <b>${s.name}</b> — <b>${s.score}점</b></li>`;

                });

            }

        }


        if (role === 'student') {

            const myData = state.students[socket.id];

            if (myData) document.getElementById('s-score').innerText = myData.score;


            const city = state.cities[state.currentCityIndex];

            const contentEl = document.getElementById('game-content');

            

            if (state.step === 'waiting') {

                document.getElementById('s-title').innerText = '선생님의 다음 지시를 기다리는 중...';

                contentEl.innerHTML = '<p>화면을 주목해주세요!</p>';

            } else if (state.step === 'location') {

                document.getElementById('s-title').innerText = `📍 [${city.season}] 도시 위치 파악하기`;

                contentEl.innerHTML = `<div style="background:#f1f5f9; padding:20px; border-radius:12px; margin:20px 0;"><strong>지리 힌트:</strong> ${city.hint}</div><p>제시된 힌트와 국기를 보고 지도에서 알맞은 국가를 찾아보세요.</p>`;

            } else if (state.step === 'packing') {

                document.getElementById('s-title').innerText = `🧳 ${city.name} (${city.climate} 기후) 가방 싸기`;

                let html = `<p>기후 특성에 알맞은 준비물을 모두 골라 담아주세요!</p><div class="options-grid">`;

                city.options.forEach(opt => {

                    const isSel = selectedBag.includes(opt) ? 'selected' : '';

                    html += `<div class="item-btn ${isSel}" onclick="toggleItem('${opt}')">${opt}</div>`;

                });

                html += `</div><button class="btn" style="width:100%; margin-top:10px;" onclick="submitBag()">가방 싸기 완료 제출</button>`;

                contentEl.innerHTML = html;

            } else if (state.step === 'result') {

                document.getElementById('s-title').innerText = `✨ ${city.name} 정답 및 해설`;

                contentEl.innerHTML = `<div style="background:#ecfdf5; border:1px solid #a7f3d0; padding:20px; border-radius:12px; margin:20px 0; color:#065f46;"><h3 style="margin-top:0; color:#047857;">정답 준비물</h3><p style="font-size:18px; font-weight:bold; color:#065f46;">${city.answer.join(', ')}</p></div><p>선생님의 설명을 들으며 기후적 원인과 인간 생활의 관계를 정리해 봅시다!</p>`;

            }

        }

    });


    function toggleItem(item) {

        const idx = selectedBag.indexOf(item);

        if (idx > -1) selectedBag.splice(idx, 1);

        else selectedBag.push(item);

        

        const btns = document.querySelectorAll('.item-btn');

        btns.forEach(b => {

            if (selectedBag.includes(b.innerText)) b.classList.add('selected');

            else b.classList.remove('selected');

        });

    }


    function submitBag() {

        if (selectedBag.length === 0) return alert('준비물을 최소 한 개 이상 골라주세요!');

        socket.emit('submit_bag', selectedBag);

        alert('답안이 제출되었습니다! 결과를 기다려주세요.');

        selectedBag = [];

    }


    socket.on('result_feedback', (res) => {

        alert(res.message);

    });

</script>
